cmake_minimum_required(VERSION 3.16)
project(zm_camera)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui)
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/vendor/extracted")
add_executable(zm_camera camera_node.cpp)
target_include_directories(zm_camera PRIVATE "${SDK}/include" ${OpenCV_INCLUDE_DIRS})
target_link_directories(zm_camera PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/runtime")
ament_target_dependencies(zm_camera rclcpp sensor_msgs)
target_link_libraries(zm_camera "${SDK}/lib/m64x86/libMVSDK.so.2.1.0.194985" ${OpenCV_LIBS})
target_link_options(zm_camera PRIVATE "-Wl,-rpath-link,${SDK}/lib/m64x86/GenICam/bin/Linux64_x64" "-Wl,-rpath-link,${CMAKE_CURRENT_SOURCE_DIR}/runtime")
