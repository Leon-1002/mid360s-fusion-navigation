cmake_minimum_required(VERSION 3.16)
project(zm_camera)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui)
set(CAMERA_SDK_ROOT "" CACHE PATH "Extracted vendor SDK root selected by build.sh")
set(CAMERA_SDK_LIBRARY "" CACHE FILEPATH "Native MVSDK shared library selected by build.sh")
set(CAMERA_GENICAM_LIB "" CACHE PATH "Native GenICam library directory")
set(CAMERA_RUNTIME_DIR "" CACHE PATH "Architecture-specific runtime aliases")
if(NOT EXISTS "${CAMERA_SDK_ROOT}/include/IMVApi.h" OR NOT EXISTS "${CAMERA_SDK_LIBRARY}")
  message(FATAL_ERROR "Run build.sh to select and validate the native camera SDK")
endif()
add_executable(zm_camera camera_node.cpp)
target_include_directories(zm_camera PRIVATE "${CAMERA_SDK_ROOT}/include" ${OpenCV_INCLUDE_DIRS})
target_link_directories(zm_camera PRIVATE "${CAMERA_RUNTIME_DIR}")
ament_target_dependencies(zm_camera rclcpp sensor_msgs)
target_link_libraries(zm_camera "${CAMERA_SDK_LIBRARY}" ${OpenCV_LIBS})
target_link_options(zm_camera PRIVATE "-Wl,-rpath-link,${CAMERA_GENICAM_LIB}" "-Wl,-rpath-link,${CAMERA_RUNTIME_DIR}")
set_target_properties(zm_camera PROPERTIES BUILD_RPATH "${CAMERA_RUNTIME_DIR};${CAMERA_GENICAM_LIB}")
